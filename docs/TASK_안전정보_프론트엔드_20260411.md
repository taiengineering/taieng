### A-2. 뉴스 그리드 API 연동

```javascript
const TAG_MAP = {
  'law':        { label: '법령 개정', cls: 'tag-law' },
  'accident':   { label: '재해 사례', cls: 'tag-safe' },
  'inspection': { label: '감독 동향', cls: 'tag-notice' },
  'penalty':    { label: '판례·과태료', cls: 'tag-guide' },
};

async function loadNews(type = 'all', btn = null) {
  if (btn) {
    document.querySelectorAll('.cat-btn').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
  }

  const grid = document.getElementById('newsGrid');
  grid.innerHTML = '<div class="text-center py-5"><div class="spinner-border text-secondary"></div></div>';

  try {
    const params = type === 'all' ? '' : `&post_type=${type}`;
    const res = await fetch(`${API}/posts?size=6${params}`);
    const data = await res.json();
    const posts = data.data || [];

    if (!posts.length) {
      grid.innerHTML = '<div class="col-12 text-center py-5 text-muted">수집된 정보가 없습니다</div>';
      return;
    }

    grid.innerHTML = posts.map(p => {
      const tag = TAG_MAP[p.post_type] || { label: '정보', cls: 'tag-guide' };
      const date = (p.published_at || p.created_at || '').slice(0, 10);
      return `
        <div class="col-lg-4 col-md-6">
          <div class="news-card" onclick="location.href='safety-news-detail.html?id=${p.id}'" style="cursor:pointer;">
            <div class="news-card-body">
              <span class="news-tag ${tag.cls}">${tag.label}</span>
              <h5>${p.title}</h5>
              <p>${(p.content || '').slice(0, 80)}...</p>
              <div class="news-meta"><span>${date}</span><span>${p.source || 'TAI'}</span></div>
            </div>
          </div>
        </div>`;
    }).join('');
  } catch(e) {
    grid.innerHTML = '<div class="col-12 text-center py-4 text-muted">데이터를 불러오는 중 오류가 발생했습니다</div>';
  }
}

// 페이지 로드 시 자동 실행
document.addEventListener('DOMContentLoaded', () => loadNews('all'));
```

### A-3. 판례 하이라이트 섹션 API 연동

```javascript
async function loadPrecHighlights() {
  try {
    // 산업안전 판례
    const r1 = await fetch(`${API}/precedents/search?query=산업안전보건&size=3`);
    const d1 = await r1.json();
    renderPrecList('precSafety', d1.data?.prec || [], false);

    // 산재 전용 판례
    const r2 = await fetch(`${API}/precedents/search?query=산재&source=sanjae&size=3`);
    const d2 = await r2.json();
    renderPrecList('precSanjae', d2.data?.prec || [], true);
  } catch(e) {
    console.error('판례 로드 실패', e);
  }
}

function renderPrecList(containerId, items, isSanjae) {
  const el = document.getElementById(containerId);
  if (!items.length) { el.innerHTML = '<p class="text-muted text-center py-3">판례 없음</p>'; return; }
  el.innerHTML = items.map(p => `
    <a href="${p.판례상세링크 || '#'}" target="_blank" class="prec-mini-card ${isSanjae ? 'sanjae' : ''} text-decoration-none">
      <div class="prec-mini-icon"><i class="fas fa-${isSanjae ? 'file-medical' : 'gavel'}"></i></div>
      <div>
        <h6>${p.사건명 || ''}</h6>
        <div class="prec-mini-meta">${p.선고일자 || ''} · ${p.법원명 || ''}</div>
      </div>
    </a>`).join('');
}
```

### B-2. 검색 로직

```javascript
let currentSource = 'all';
let currentPage = 1;

function setSource(src, btn) {
  currentSource = src;
  document.querySelectorAll('.cat-btn').forEach(b => b.classList.remove('active'));
  btn.classList.add('active');
  searchPrec();
}

async function searchPrec(page = 1) {
  currentPage = page;
  const query = document.getElementById('precQuery').value.trim() || '산업안전';
  const sourceParam = currentSource === 'sanjae' ? '&source=sanjae' : '';
  const url = `${API}/precedents/search?query=${encodeURIComponent(query)}&page=${page}&size=10${sourceParam}`;

  showLoading();
  try {
    const res = await fetch(url);
    const data = await res.json();
    renderResults(data.data?.prec || [], data.data?.totalCnt || 0);
  } catch(e) {
    showError();
  }
}

function renderResults(items, total) {
  const el = document.getElementById('precResults');
  document.getElementById('resultCount').textContent = `${total.toLocaleString()}건`;

  if (!items.length) {
    el.innerHTML = '<div class="text-center py-5 text-muted">검색 결과가 없습니다</div>';
    return;
  }

  el.innerHTML = items.map(p => {
    const isSanjae = p.데이터출처명?.includes('산재');
    return `
    <a href="${p.판례상세링크 || '#'}" target="_blank" class="prec-result-card ${isSanjae ? 'sanjae' : ''} text-decoration-none">
      <div class="d-flex justify-content-between align-items-start gap-3">
        <div>
          <div class="prec-source-badge ${isSanjae ? 'badge-sanjae' : 'badge-law'}">
            ${isSanjae ? '산재 전용' : '산업안전'}
          </div>
          <div class="prec-title">${p.사건명 || ''}</div>
          <div class="prec-meta">${p.사건번호 || ''} · ${p.법원명 || ''} · ${p.선고일자 || ''}</div>
        </div>
        <div class="prec-verdict ${p.판결유형 === '기각' ? 'verdict-red' : 'verdict-green'}">
          ${p.판결유형 || ''}
        </div>
      </div>
    </a>`;
  }).join('');
}
```
